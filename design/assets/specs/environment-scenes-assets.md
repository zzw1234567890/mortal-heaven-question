## 缩放落盘 SOP

from PIL import Image
# 场景图：2560×1440 → 1920×1080（等比 0.75，无裁切）
src = Image.open(r'<候选件路径>/main_menu_cloud_sea.png').convert('RGB')

    r'E:/mortal-heaven-question/assets/environments/backgrounds/menu/main_menu_cloud_sea.png')
